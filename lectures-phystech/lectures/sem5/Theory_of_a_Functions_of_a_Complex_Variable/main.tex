\lecture{1}

\section{Введение}
Вспомним, как мы вводили \(\Cm\) --- это расширение \(\R\) единственным числом \(i\): \(\Cm = \R[i]\). Соответственно, \(\forall z \in \Cm: z = \alpha + \beta i\). Перемножение и сложение комплексных чисел работают следующим образом:
\[(\alpha + \beta i) + (\gamma + \delta i) = (\alpha + \gamma) + (\beta + \delta)i\]
\[(\alpha + \beta i) \cdot (\gamma + \delta i) = (\alpha \gamma - \beta \delta) + (\alpha \delta + \beta \delta)i\]
\subsection{Напоминание}
Пусть \(a = \alpha + \beta i\). Напомним несколько определений и фактов:
\begin{enumerate}
    \item \(\overline{a} = \alpha - \beta i\)
    \item \(\Re a = \alpha, \Im a = \beta\)
    \item \(|a| = \sqrt{\alpha^2 + \beta^2} = a\overline{a}\)
    \item \(|ab| = |a||b|\)
    \item \(\overline{a + b} = \overline{a} + \overline{b}\)
    \item \(\overline{a \cdot b} = \overline{a} \cdot \overline{b}\)
    \item \(-|a| \le \Re a \le |a|\) --- следует из определения \(|a|\).
    \item Неравенство треугольника: \(|a + b| \le |a| + |b|\) или \(|a| - |b| \le |a - b|\)
\end{enumerate}

Вспомним также тригонометрическую запись комплексного числа:
\(0 \ne a = r(\cos \phi + i \sin \phi)\). В таком случае \(|a| = r\), \(\phi \in \operatorname{Arg}\{a\} = \{\operatorname{arg}a + 2\pi k\}_{k \in \Z}\), где \(\operatorname{arg}a \in (-\pi, \pi)\).

\begin{exercise}
    \(\operatorname{Arg}\{a_1a_2\} = \operatorname{Arg}\{a_1\} + \operatorname{Arg}\{a_2\}\), что равносильно:
    \(\operatorname{arg}\{a_1a_2\} = \operatorname{arg}\{a_1\} + \operatorname{arg}\{a_2\} \mod 2\pi\)
\end{exercise}

\subsection{Простейшие определения}

Пусть теперь \(z^n = a = r(\cos \phi + i\sin \phi), a \ne 0\). Будем говорить, что \(z \in \{\sqrt[n]{a}\} = \left\{ r^{\frac{1}{n}}\left( \cos \frac{\phi + 2\pi k}{n} + i\sin \frac{\phi + 2\pi k}{n} \right) \right\}_{k = 0}^{n - 1}\).

\begin{definition}
    \(O_r(a) = \{z: |z - a| < r\}, \overline{O}_r(a) = \{z: |z - a| \le r\}\)
\end{definition}

\begin{definition}
    \(\lim_{n \ra \infty} a_n = a\) тогда и только тогда, когда \(\forall \epsilon > 0 \exists N: \forall n > N: a_n \in O_\epsilon(a)\)
\end{definition}

\begin{note}
    \[\begin{array}{c}
        \Re (a_n - a) \le \\
        \Im (a_n - a) \le
    \end{array} |a_n - a| \le |\Re (a_n - a)| + |\Im (a_n - a)|\]
    Поэтому сходимость последовательности комплексных чисел равносильна покоординатной сходимости
\end{note}

\begin{note}
    Точно так же, как и в вещественном случае можно говорить о фундаментальности, теореме Больцано-Вейерштрасса, полноте \(\Cm\).
\end{note}

\begin{definition}
    Будем говорить, что \(\lim_{n \ra \infty} a_n = \infty\), если \(\forall \epsilon > 0 \exists N: \forall n > N: |a_n| > \frac{1}{\epsilon}\).
\end{definition}

Соответственно, расширим принцип Больцано-Вейерштрасса:

\begin{proposition}
    \(\forall \{a_n\} \exists \{n_k\}: a_{n_k}\) сходится в \(\overline{\Cm} = \Cm \cup \{\infty\}\)
\end{proposition}

\section{Непрерывность и \(\Cm\)-дифференцируемость}

Далее будем говорить, что \(z = x + iy\).

\begin{definition}
    Пусть \(f: \stackrel{\circ}{O}_r(a) \ra \Cm\). Будем говорить, что \(\lim_{z \ra a} f(z) = A\), если \(\forall \epsilon > 0 \exists \delta > 0: \forall z \in \stackrel{\circ}{O}_\delta(a): f(z) \in O_\epsilon(A)\)
\end{definition}

\begin{definition}
    \(f: O_r(a) \ra \Cm\) называется непрерывной в точке \(a\), если \(\lim_{z \ra a} f(z)\)
\end{definition}

\begin{note}
    \(f\) непрерывна в \(a \Ra \Re f, \Im f, |f|\) --- непрерывны в точке \(a\).
\end{note}

\begin{definition}
    \(f: O_r(a) \ra \Cm\) называется дифференцируемой в точке \(a\) (\(\Cm\)-дифференцируемой в точке \(a\)), если \(\exists \lim_{z \ra a} \frac{f(z) - f(a)}{z - a} = A \in \Cm\)
\end{definition}

\begin{theorem}
    Пусть \(f: O_r(a) \ra \Cm, \Re f = u, \Im f = v\). Тогда она является \(\Cm\)-дифференцируемой в точке \(a \Lra u, v\) дифференцируемы в вещественном смысле в точке \(a\) и выполнено условие Коши:
    \begin{equation*}
        \begin{cases*}
            u_x = v_y \\
            u_y = -v_x \\
        \end{cases*}
    \end{equation*}
\end{theorem}
\begin{proof}
    Положим \(\Delta f = f(z) - f(a), \Delta z = z - a\). В таком случае \(\Cm\)-дифференцируемость функции \(f\) равносильна: \(\Delta f = A \Delta z + o(|\Delta z|)\). Положим \(A = \gamma + \delta i\), тогда равносильное условие \(\Cm\)-дифференцируемости функции \(f\) можно переписать так:
    \[\Delta f = (\gamma + \delta i)(\Delta x + i \Delta y) + o(|\Delta z|) = (\gamma + \delta i)\Delta x + (-\delta +\gamma i)\Delta y + o(|\Delta z|)\]
    
    Теперь, \(u, v\) дифференцируемы в точке \(a\) тогда и только тогда, когда:
    \[\left\{\begin{array}{l}
        \Delta u = u_x \Delta x + u_y \Delta y + o(|\Delta z|) \\
        \Delta v = v_x \Delta x + v_y \Delta y + o(|\Delta z|)
    \end{array}\right.\]
    
    В таком случае, \(\Delta f = \Delta u + i\Delta v = (u_x + iu_y)\Delta x + (u_y + iv_y)\Delta y + o(|\Delta z|)\).

    Теперь, положив \(\gamma = u_x = v_y, \delta = v_x = -u_y\), получаем требуемое.
\end{proof}

\begin{example}[Условие Коши существенно]
    Рассмотрим функцию \(f(z) = |z|^2 = x^2 + y^2 + i \cdot 0\). Запишем условие Коши:
    \[\left\{\begin{array}{l}
        2x = u_x \stackrel{?}{=} v_y = 0 \\
        2y = u_y \stackrel{?}{=} -v_x = 0 \\
    \end{array}\right.\]
    Получаем, что \(f\) дифференцируема только в нуле.
\end{example}

\begin{note}
    Сумма, произведение \(\Cm\)-дифференцируемых функций дифференцируема
\end{note}

\begin{definition}
    Функция \(f\) называется голоморфной на \(E \subset \Cm\), если она \(\Cm\)-дифференцируема в открытом множестве, содержащем множество \(E\)
\end{definition}

\begin{note}
    Таким образом \(f(z) = |z|^2\) не является голоморфной ни в какой точке.
\end{note}

\begin{definition}
    Множество \(E \subset \Cm\) (или \(\Cm\)) называется связным, если не существует двух открытых множеств \(G_1, G_2\), таких, что:
    \begin{enumerate}
        \item \(E \subset G_1 \cup G_2\)
        \item \(E \cap G_1 \cap G_2 = \emptyset\)
        \item \(E \cap G_1, E \cap G_2 \ne \emptyset\)
    \end{enumerate}
\end{definition}

\begin{note}
    Если в определении выше множество \(E\) открыто, то пункт 2 можно заменить на \(G_1 \cap G_2 = \emptyset\)
\end{note}

\begin{proposition}
    Пусть \(E\) --- промежуток (отрезок, полуинтервал, интервал, луч или прямая). Тогда \(E\) --- связное множество.
\end{proposition}
\begin{proof}
    От противного. Пусть \(\exists G_1, G_2\) --- открытые в \(\Cm\), что выполнены свойства 1-3. Тогда \(\exists a \ne b \in E: a \in G_1, b \in G_2\). Таким образом, мы свели задачу к случаю отрезка \(E_1 = [a, b]\). Рассмотрим последовательность вложенных отрезков (будем делить каждый предыдущий пополам и выбирать подходящий) \([a_n, b_n]\), которые обладают свойством ''отрезок нельзя покрыть одним из множеств \(G_1, G_2\)''. В таком случае \(a_n \ra c, b_n \ra c\). Заметим, что \(c \in G_1 \cup G_2\). Пусть Б.О.О, \(c \in G_1\). Тогда \(\exists r > 0: O_r(c) \subset G_1 \Ra \) для достаточно большого \(n\) мы смогли покрыть отрезок \([a_n, b_n]\) множеством \(G_1\), противоречие.
\end{proof}

\begin{proposition}
    Пусть \(D\) --- открытое непустое множество в \(\overline{\Cm}\). Тогда \(D\) связно тогда и только тогда, когда \(\forall z_1, z_2 \in D \exists\) ломаная их соединяющая. Более того, ребра ломаной можно считать параллельными осям координат.
\end{proposition}
\begin{proof}
    \begin{enumerate}
        \item[\(\Ra\)] От противного. Рассмотрим точки \(z_1, z_2\) и множества \(G_1 = \{z \in D: z_1\) можно соединить с \(z\) ломаной из условия теоремы \(\}\), \(G_2 = D \setminus G_1\) Эти множества будут открытыми. Действительно, \(G_1\) открыто так как утверждение теоремы справедливо для любого открытого шара, аналогично для \(G_2\) (если до точки можно дойти по ломаной, то и до любой другой из ее окрестности тоже). Кроме того, \(G_1, G_2 \ne \emptyset\), так как по предположению \(z_1 \in G_1, z_2 \in G_2\). Второе свойство выполнено, так как \(G_1 \cap G_2 = \emptyset\). Но тогда получили противоречие со связностью
        \item[\(\La\)] (Здесь я написал свое доказательство) Пусть \(D\) не связно. Тогда рассмотрим множества \(G_1, G_2\) из определения и точки \(z_1, z_2\), между которыми существет путь \(\gamma_{z_1, z_2}\). Заметим, что \(\gamma_{z_1, z_2}: [0, 1] \ra \Cm\) --- непрерывное отображение, поэтму \(\gamma_{z_1, z_2}^{-1}(\text{открытое}) = \text{открытое}\). Поэтому \(\gamma_{z_1, z_2}^{-1}(G_1), \gamma_{z_1, z_2}^{-1}(G_2)\) --- открытые множества (на прямой), удовлетворяющие условиям 1-3 и покрывающие отрезок \([0, 1]\), что приводит нас к противоречию.
    \end{enumerate}
\end{proof}

\begin{definition}
    \(D \ne \emptyset\) --- область в \(\overline{\Cm}\), если \(D\) открыто и связно
\end{definition}

\begin{theorem}
    Пусть \(f\) голоморфна в области \(D \subset \Cm\). Тогда \(f' \equiv 0 \Ra f \equiv const\)
\end{theorem}
\begin{proof}
    От противного. Пусть \(f(z_1) \ne f(z_2)\). Так как \(D\) --- область, то \(\exists \gamma_{z_1, z_2}\) --- ломаная, соединяющая точки \(z_1, z_2\) со звеньями, параллельными осям координат. Запишем \(f(x, y) = u(x, y) + i v(x, y)\). Заметим, что вдоль каждого звена \(f(x, y)\) является константой из вещественного случая (т.к. \(u(x, \cdot), v(x, \cdot), u(\cdot, y), v(\cdot, y)\) постоянны). В таком случае \(f(z_1) = f(z_2)\), противоречие
\end{proof}

\begin{proposition}
    Пусть \(f: O_r(z_0) \ra O_\rho(w_0)\) голоморфна в \(O_r(z_0), f(z_0) = w_0, g\) голоморфна в \(O_\rho(w_0)\). Тогда \(h = g \circ f\) тоже голоморфна в \(O_r(z_0)\), причем \(h'(z_0) = g'(w_0)f'(z_0)\).
\end{proposition}
\begin{proof}
    Положим \(w = f(z), \Delta z = z - z_0, \Delta w = w - w_0\). В таком случае: \(\Delta w = f'(z_0)\Delta z + o(\Delta z)\). Тогда: \(g(w) - g(w_0) = g'(w_0)\Delta w + \Delta w \eta(\Delta w)\), где \(\lim_{\Delta w \ra 0}\eta(\Delta w) = 0\). В таком случае:
    \[\frac{h(z) - h(z_0)}{z - z_0} = \frac{g(w) - g(w_0)}{\Delta z} = g'(w_0) \frac{\Delta w}{\Delta z} + \frac{\Delta w}{\Delta z}\eta(\Delta w) = (*)\]
    Получаем, что \(\lim_{z \ra z_0} (*) = g'(w_0)f(z_0)\).
\end{proof}

\begin{theorem}[Об обратной функции]
    Пусть \(f\) голоморфна в \(D\), \(f'\) непрерывна в области \(D\) (докажем позднее, что это следует из голоморфности), \(z_0 \in D, f(z_0) = w_0, f'(z_0) \ne 0\). Тогда \(\exists U, V\) --- окрестности точек \(z_0, w_0\), такие, что \(f\) --- биекция \(U\) на \(V\), а \(g = f^{-1}\) голоморфна в \(V\) и \(g'(w_0) = \frac{1}{f(z_0)}\)
\end{theorem}
\begin{proof}
    \(f'\) непрерывна \(\Ra |f'|\) тоже будет непрерывной. Тогда \(\exists O_r(z_0): |f'(z)| > 0 \forall z \in O_r(z_0)\). Рассмотрим \(f\) как \(f: \R^2 \ra \R^2\) и посчитаем якобиан:
    \[J_f = \left| \begin{array}{cc}
        u_x & u_y \\
        v_x & v_y
    \end{array} \right| = u_xv_y - u_yv_x = u_x^2 + u_y^2 = |f'(z)|^2 > 0\]
    В таком случае по теореме об обратной функции: \(\exists U, V\) --- окрестности точек \(z_0, w_0\), такие, что \(f\) --- биекция \(U\) на \(V\), а \(g = f^{-1}\) дифференцируема в \(V\) и \(g'(w_0) = \frac{1}{f(z_0)}\). Докажем голоморфность \(g\):

    Так как \(f\) \(\Cm\)-дифференцируема в точке \(z_0\),
    \[
        \Delta w=f'(z_0)\Delta z+o(|\Delta z|).
    \]
    Поскольку \(f'(z_0)\ne 0\), при достаточно малых \(\Delta z\)
    \[
        |\Delta w|\ge c|\Delta z|,  c>0,
    \]
    поэтому \(o(|\Delta z|)=o(|\Delta w|)\). Следовательно,
    \[
        \Delta z=\frac{1}{f'(z_0)}\Delta w+o(|\Delta w|).
    \]
    Но \(\Delta z=g(w)-g(w_0)\), \(\Delta w=w-w_0\). Значит,
    \[
        g(w)-g(w_0)=\frac{1}{f'(z_0)}(w-w_0)+o(|w-w_0|).
    \]
    Это и означает, что \(g\) \(\Cm\)-дифференцируема в точке \(w_0\), причём
    \[
        g'(w_0)=\frac{1}{f'(z_0)}=\frac{1}{f'(g(w_0))}.
    \]
    Так как это верно для каждой точки \(w_0\in V\), функция \(g\) голоморфна в \(V\).
\end{proof}

\begin{definition}
    Голоморфная функция \(f\) в области \(D\) называется однолистной, если \(\forall z_1 \ne z_2 \in D: f(z_1) \ne f(z_2)\)
\end{definition}

\begin{definition}
    Голоморфная функция \(f\) в области \(D\) называется локально однолистной, если \(\forall z \in D \exists O(z): f\) однолистна в \(O(z)\).
\end{definition}

\begin{example}
    \(f(z)=z^2\) является локально однолистной, но не однолистной. Действительно,
    \[
        f(z)=z^2=(-z)^2=f(-z),
    \]
    поэтому на \(\Cm\) функция \(z^2\) не однолистна. Если \(z_0\ne 0\), то
    \[
        f'(z_0)=2z_0\ne 0,
    \]
    и по теореме об обратной функции \(f\) локально однолистна в окрестности точки \(z_0\). В точке \(z_0=0\) локальной однолистности нет: в любой её окрестности найдутся \(z\) и \(-z\), для которых
    \[
        z^2=(-z)^2.
    \]

    Итак, \(f(z)=z^2\) локально однолистна в \(\Cm\setminus\{0\}\), но не однолистна на всём \(\Cm\).
\end{example}

\subsection{Степенные ряды}

\begin{definition}
    Пусть \(E \subset \Cm, f_n: E \ra \Cm\). Будем говорить, что \(f_n \ra f\) (сходится поточечно), если \(\forall z \in E: f_n(z) \ra f(z)\).
\end{definition}

\begin{definition}
    Пусть \(E \subset \Cm, f_n: E \ra \Cm\). Будем говорить, что \(f_n \rightrightarrows_E f\) (сходится равномерно), если \(\forall \epsilon > 0\exists N: \forall n > N \forall z \in E: |f_n(z) - f(z)| < \epsilon\).
\end{definition}

\begin{note}
    \(f_n \rightrightarrows_E f \Lra \forall \epsilon > 0\exists N: \forall n, m > N \forall z \in E: |f_n(z) - f_m(z)| < \epsilon\)
\end{note}

\begin{definition}
    Будем говорить, что ряд \(S = \sum_{n = 0}^\infty f_n\) сходится равномерно, если \(S_n = \sum_{k = 0}^n f_k \rightrightarrows S\)
\end{definition}

\begin{note}
    Аналогично вещественному случаю верен признак сходимости Вейерштрасса, а именно: если \(\sum_{n = 1}^\infty \alpha_n < +\infty, |f_n| \le \alpha_n\), то \(\sum_{n = 1}^\infty f_n \rightrightarrows\)
\end{note}

\begin{theorem}
    Пусть дан ряд \(S(z) = \sum_{n = 0}^\infty a_nz^n, a_n \in \Cm\). Положим \(R = \frac{1}{\limsup_{n \ra \infty} \sqrt[n]{|a_n|}} \in [0, +\infty]\). Тогда:
    \begin{enumerate}
        \item При \(|z| \le r < R\) ряд \(S\) сходится равномерно
        \item При \(|z| > R\) ряд расходится
        \item При \(|z| < R\) функция \(S(z)\) голоморфна, причем \(S'(z) = \sum_{n = 1}^\infty na_nz^{n - 1}\).
    \end{enumerate}
\end{theorem}
\begin{proof}
    \begin{enumerate}
        \item Докажем при \(R > 0\). Пусть \(\rho \in (r, R)\). Тогда \(\limsup \sqrt[n]{|a_n|} = \frac{1}{R} < \frac{1}{\rho}\). Это означает, что \(\exists N: \forall n > N: |a_n| \le \frac{1}{\rho^n}\) и тогда \(|a_nz^n| \le \left( \frac{r}{\rho} \right)^n\). Так как \(\frac{r}{\rho} < 1\), применяя признак Вейерштрасса, получаем, что \(S_n(x) \rightrightarrows S(x)\).
        \item Пусть \(|z| > R\). Заметим, что тогда \(\exists \{n_k\} \uparrow\), такая, что \(\sqrt[n_k]{|a_{n_k}|} \ra \frac{1}{R}\). Но в таком случае \(|a_{n_k}z^{n_k}| \ge 1\), т.е. слагаемые ряда не сходится к 0, т.е. он расходится.
        \item Рассмотрим \(g(z) = \sum_{n = 1}^\infty na_nz^n\). Заметим, что радиус сходимости данного ряда равен: \[\frac{1}{\limsup_{n \ra \infty}\sqrt[n]{n + 1|a_{n + 1}|}} = \frac{1}{\lim_{n \ra \infty} \sqrt[n]{n + 1}\cdot\limsup_{n \ra \infty}\sqrt[n]{|a_{n + 1}|}} = \frac{1}{\limsup_{n \ra \infty} \sqrt[n]{|a_n|}} = R\]
        Положим \(Q_n(z) = \sum_{k = n + 1}^\infty a_kz^k\). Тогда:
        \[\frac{S(z) - S(z_0)}{z - z_0} - g(z_0) = \frac{S_n(z) - S_n(z_0)}{z - z_0} + \frac{Q_n(z) - Q_n(z_0)}{z - z_0} - g(z_0) =\]
        \[\underbrace{\frac{S_n(z) - S_n(z_0)}{z - z_0} - S_n'(z_0)}_{I_1} + \underbrace{\frac{Q_n(z) - Q_n(z_0)}{z - z_0}}_{I_2} + \underbrace{S_n'(z_0) - g(z_0)}_{I_3}\]

        Зафиксируем \(r: |z_0| < r < R\). Тогда при \(|z| \le r\):
        \[
        |I_2|
        =
        \left|
        \sum_{k=n+1}^\infty a_k \frac{z^k-z_0^k}{z-z_0}
        \right|
        \le
        \sum_{k=n+1}^\infty k|a_k|r^{k-1}
        \ra 0 \quad (n \ra \infty),
        \]
        так как \(r < R\). Далее,
        \[
        I_3
        =
        S_n'(z_0)-g(z_0)
        =
        -\sum_{k=n+1}^\infty k a_k z_0^{k-1}
        \ra 0 \quad (n \ra \infty),
        \]
        поскольку \(|z_0| < R\). Наконец, при фиксированном \(n\)
        \[
        I_1
        =
        \sum_{k=1}^n a_k\left(\frac{z^k-z_0^k}{z-z_0}-kz_0^{k-1}\right)
        \ra 0 \quad (z \ra z_0),
        \]
        так как это конечная сумма. Следовательно, \(\forall \epsilon > 0\) выберем \(n\) так, чтобы
        \[
        |I_2| < \epsilon,\qquad |I_3| < \epsilon,
        \]
        а затем \(\delta > 0\) так, чтобы при \(0 < |z-z_0| < \delta\) было
        \(
        |I_1| < \epsilon.
        \). Тогда
        \(
        \left|
        \frac{S(z)-S(z_0)}{z-z_0}-g(z_0)
        \right|
        \le |I_1|+|I_2|+|I_3|
        < 3\epsilon
        \). Значит,
        \[
        \lim_{z \ra z_0}\frac{S(z)-S(z_0)}{z-z_0}=g(z_0),
        \Ra
        S'(z_0)=g(z_0)=\sum_{n=1}^\infty n a_n z_0^{n-1}.
        \]
        В силу произвольности \(z_0\) при \(|z_0|<R\), функция \(S\) голоморфна при \(|z|<R\), причём
        \[
        S'(z)=\sum_{n=1}^\infty n a_n z^{n-1}.
        \]
    \end{enumerate}
\end{proof}

\begin{note}
    Заметим что в теореме выше выполнено \(S^{(n)}(z) = n!a_n + \sum_{k = 1}^\infty b_kz^k\). В таком случае \(a_n = \frac{S^{(n)}(0)}{n!}\).
\end{note}

\begin{note}[Свойства экспоненты]

        \item \((e^z)' = e^z\)
        \item \(e^0 = 1\)

        \item \(e^{z_1 + z_2} = e^{z_1}e^{z_2}\)

        \item[6.] \(|e^{iy}|^2 = e^{iy}\overline{e^{iy}} = e^{iy}e^{-iy} = 1\).
    \end{enumerate}
\end{proof}

\subsection{Кривые в \(\Cm\)}

\begin{definition}
    Уравнение кривой \(\gamma\) --- непрерывная функция \(z: [\alpha, \beta] \ra \Cm\).
\end{definition}

\begin{note}
    Формально, кривая \(\gamma\) --- класс эквивалентности путей \(z\), тогда как \(z\) --- один из его представителей.
\end{note}

Мы будем рассматривать \(z\), как пару непрерывных функций \(x, y: [\alpha, \beta] \ra \R\), и \(z = x(t) + iy(t)\).

\begin{note}
    Образ кривой, а именно \(z([\alpha, \beta]) = \{z: z = x(t) = iy(t), t \in [\alpha, \beta]\}\) является компактным линейно связным множеством. Будем говорить, что \(z(\alpha)\) --- начало кривой, а \(z(\beta)\) --- конец. В случае, если \(z(\alpha) = z(\beta)\), будем говорить, что кривая замкнута.
\end{note}

\begin{definition}
    Кривая \(\gamma\) называется жордановой, если \(\exists z: z(t_1) = z(t_2) \Lra t_1 = t_2\).
\end{definition}

\begin{definition}
    Кривая \(\gamma\) называется замкнутой жордановой кривой, если \(\exists z: z(t_1) = z(t_2) \Lra t_1 = t_2 \forall t_1, t_2 \in [\alpha, \beta)\) и \(z(\alpha) = z(\beta)\)
\end{definition}

\begin{note}
    Замкнутая жордановая кривая не является жордановой кривой.
\end{note}

\begin{definition}
    По кривой \(\gamma\) построим кривую \(-\gamma\) следующим образом: для любого представителя \(z\) кривой \(\gamma\) рассмотрим \(\tilde z(t) = z(-t)\), при этом \(\tilde z: [-\beta, -\alpha]\).
\end{definition}

\begin{definition}
    Кривая \(\gamma\) называется гоадкой, если \(\exists\) параметризация кривой, такая, что \(z = x + iy, x, y \in C^1[\alpha, \beta]\), причем \(z'(t) = \dot x(t) + i\dot y(t) \ne 0\)
\end{definition}

\begin{definition}
    Кривая \(\gamma\) называется кусочно-гладкой, если \(\exists T = \alpha = t_0 < t_1 < \dots < t_{n - 1} < t_n = \beta\) --- разбиение отрезка \([\alpha, \beta]\) и параметризация \(z\), такие, что \(z|_{[t_k, t_{k + 1}]}\) является гладкой
\end{definition}

\begin{definition}
    \(\operatorname{Length} \gamma = \sup_T \sum_{k = 0}^n |z(t_k - z(t_{k - 1}))|\)
\end{definition}

\subsection{Определение интеграла вдоль кривой}

Пусть \(D\) --- область в \(\Cm, \gamma \subset D, \gamma\) --- кусочно-гладкая кривая, \(f\) непрерывна в области \(D\). Пусть \(T, \tau\) --- отмеченное разбиение отрезка \([\alpha, \beta]\), то есть \(\tau_k \in [t_{k - 1}, t_k]\). 

\begin{definition}
    \[\int_\gamma fdz = \lim_{|T| \ra 0} \sum_{} f(z(\tau_k))(z(t_k) - z(t_{k - 1}))\]
\end{definition}
\begin{definition}
    \[\int_\gamma f|dz| = \lim_{|T| \ra 0} \sum_{} f(z(\tau_k))|z(t_k) - z(t_{k - 1})|\]
\end{definition}

Пусть \(f=u+iv\), где \(u=\Re f,\ v=\Im f\), и \(dz=dx+i\,dy\). Тогда

\[
\int_\gamma f\,dz
=
\int_\gamma (u+iv)\,dz
=
\int_\gamma u\,dz+i\int_\gamma v\,dz.
\]

Распишем \(dz=dx+i\,dy\):

\[
\int_\gamma u\,dz
=
\int_\gamma u\,dx+i\int_\gamma u\,dy,
\qquad
\int_\gamma v\,dz
=
\int_\gamma v\,dx+i\int_\gamma v\,dy.
\]

Следовательно,

\[
\int_\gamma f\,dz
=
\int_\gamma (u\,dx-v\,dy)
+
i\int_\gamma (v\,dx+u\,dy).
\]

То есть

\[
\Re\int_\gamma f\,dz
=
\int_\gamma (u\,dx-v\,dy),
\qquad
\Im\int_\gamma f\,dz
=
\int_\gamma (v\,dx+u\,dy).
\]

Если кривая задана параметризацией \(z(t)=x(t)+iy(t)\), \(t\in[\alpha,\beta]\), то

\[
\int_\gamma f\,dz
=
\int_\alpha^\beta
\bigl(u(x(t),y(t))x'(t)-v(x(t),y(t))y'(t)\bigr)\,dt
+
i\int_\alpha^\beta
\bigl(v(x(t),y(t))x'(t)+u(x(t),y(t))y'(t)\bigr)\,dt.
\]

Отдельно для интеграла по длине дуги:

\[
\int_\gamma f\,|dz|
=
\int_\gamma u\,|dz|
+
i\int_\gamma v\,|dz|,
\]

поэтому

\[
\Re\int_\gamma f\,|dz|
=
\int_\gamma u\,ds,
\qquad
\Im\int_\gamma f\,|dz|
=
\int_\gamma v\,ds.
\]

\begin{note}[Свойства интеграла вдоль кривой]
    \begin{enumerate}
        \item \textbf{Линейность}. Пусть \(f = a f_1 + b f_2\). Тогда \(\int_\gamma fdz = a\int_\gamma f_1dz + b\int_\gamma f_2dz\), \(\int_\gamma f|dz| = a\int_\gamma f_1|dz| + b\int_\gamma f_2|dz|\)
        
        \item \textbf{Аддитивность}. Пусть \(\gamma = \gamma_1 + \gamma_2\). Тогда \(\int_\gamma = \int_{\gamma_1} + \int_{\gamma_2}\)
        
        \item \textbf{Неравенство треугольника}. \(\left|\int_\gamma fdz\right| \le \int_\gamma |f|dz \le \max_\gamma|f|\int_\gamma |dz| = \max_\gamma |f| \operatorname{Length} \gamma\).
    \end{enumerate}
\end{note}

\begin{definition}
    Пусть \(f: D \ra \Cm\), где \(D\) --- область в \(\Cm\). Будем говорить, что \(f\) имеет первообразную в \(D\), если \(\exists F\) --- голоморфная функция в области \(D\), такая, что \(F'(z) = f(z) \forall z \in D\).
\end{definition}

\begin{note}
    В условиях определения выше \(f(z)dz = dF\), то есть \(f(z)dz\) --- полный дифференциал.
\end{note}

\begin{theorem}
    Пусть \(f\) --- непрерывна в \(D\)
    \begin{enumerate}
        \item Если \(f(z)dz\) --- полный дифференциал, то \(\forall \) замкнутой кусочно-гладкой кривой \(\gamma \subset D: \int_\gamma fdz = 0\)
        \item Если \(\forall \gamma\) --- замкнутой ломаной \(\gamma \subset D\) верно, что \(\int_\gamma fdz = 0\), то \(f(z)dz = dF\), т.е. \(f(z)dz\) --- поный дифференциал.
    \end{enumerate}
\end{theorem}
\begin{proof}
    \begin{enumerate}
        \item Тогда \(f(z) = F'(z)\). В таком случае:
        \[\int_\gamma f(z)dz = \int_\alpha^\beta f(z(t))z'(t)dt = \int_\alpha^\beta\frac{d}{dt}F(z(t))dt = F(z(\beta)) - F(z(\alpha)) = 0\]
        \item Зафиксируем \(a \in D\). Тогда положим \(F(z) = \int_\gamma f(\zeta)d\zeta\), где \(\gamma\) --- ломаная с концами \(a, b\). Заметим, что определние корректно, т.е. \(\int_{\gamma_1} f(\zeta)d\zeta = \int_{\gamma_2} f(\zeta)d\zeta\) (так как ианче \(\int_{\gamma_1 - \gamma_2} f(\zeta)d\zeta \ne 0\)). Рассмотрим \(z_0 \in D\) и \(O_\delta(z_0) \subset D\). Пусть теперь \(z \in \dot O_\delta(z_0)\). Заметим, что тогда \(F(z) - F(z_0) = \int_{[z_0, z]}f(\zeta)d\zeta\) (достаточно рассмотреть интеграл вдоль ломаной \([a, z_0] \cup [z_0, z]\)). Теперь, для доказательства голоморфности \(F\), рассмотрим отношение (заметим, что \(\frac{1}{z - z_0}\int_{[z_0, z]}f(z_0)d\zeta f(z_0)\)):
        \[\frac{F(z) - F(z_0)}{z - z_0} - f(z_0) = \frac{1}{z - z_0}\int_{[z_0, z]} (f(\zeta) - f(z_0))d\zeta\]
        Зафиксируем \(\epsilon > 0\). Будем считать, что в \(\dot O_\delta(z_0)\) верно: \(|f(z_1) - f(z_0)| \le \epsilon\) (иначе в силу непрерывности сузим область \(\delta\)). Тогда по неравенству треугольника:
        \[\left| \frac{F(z) - F(z_0)}{z - z_0} - f(z_0) \right| = \left| \frac{1}{z - z_0}\int_{[z_0, z]} (f(\zeta) - f(z_0))d\zeta \right| = \frac{1}{|z - z_0|}\left|\int_{[z_0, z]} (f(\zeta) - f(z_0))d\zeta \right|\le\]
        \[\le \frac{1}{|z - z_0|}\int_{[z_0, z]} |f(\zeta) - f(z_0)|d\zeta \le \frac{1}{|z - z_0|}\int_{[z_0, z]} \epsilon d\zeta \le \epsilon\]
        Получили, что \(F\) голоморфна и \(F' = f\).
    \end{enumerate}
\end{proof}

\begin{example}
    Рассмотрим \(f(z) = z^n\). Заметим, что при \(n \ne -1\) верно: \(F(z) = \frac{z^{n + 1}}{n + 1}\). Попробуем понять, что происходит в случае \(n = -1\) в области \(\Cm \setminus \R_-\):

    В случае \(n=-1\) функция \(f(z)=z^{-1}=\frac1z\). В области \(\Cm \setminus \R_-\) рассмотрим ветвь логарифма
\[
    F(z)=\ln z=\ln|z|+i\arg z,\qquad \arg z\in(-\pi,\pi).
\]
Она голоморфна в \(\Cm \setminus \R_-\), причём
\[
    F'(z)=\frac1z.
\]
Значит, \(\frac1z\) имеет первообразную в \(\Cm \setminus \R_-\). Поэтому для любой замкнутой кусочно-гладкой кривой \(\gamma\subset \Cm \setminus \R_-\)
\[
    \int_\gamma \frac{dz}{z}=0.
\]
Однако в \(\Cm\setminus\{0\}\) глобальной первообразной уже нет: для окружности \(\gamma(t)=e^{it}, t\in[0,2\pi]\),
\[
    \int_\gamma \frac{dz}{z}
    =
    \int_0^{2\pi}\frac{ie^{it}}{e^{it}}\,dt
    =
    2\pi i\ne 0.
\]
\end{example}

\begin{lemma}[Гурса]
    Пусть \(f\) голоморфна в области \(D\). Тогда для любого треугольника \(\Delta: \overline{\Delta} \subset D\) выполнено:
    \[\int_{\partial \Delta} fdz = 0\]
\end{lemma}
\begin{proof}
    Пусть \(\Delta\) --- треугольник со свойством из условия. Разобьем его средними линиями на 4 треугольника. В таком случае, если \(I\) --- исходный интеграл, то \(\int_{\partial \Delta} = \int{\partial \Delta_1} + \int{\partial \Delta_2} + \int{\partial \Delta_3} + \int{\partial \Delta_4}\), где \(\Delta_j\) --- соответстующий треугольник из разбиения. Тогда из четырех полученных треугольников существует такой (Б.О.О с индексом 1), что \(\left|\int_{\partial \Delta_1} fdz\right| \ge \frac{\left| \int_{\partial \Delta} fdz\right|}{4}\). Рассмотрим \(\Delta_i\) --- последовательность вложенных треугольников, построенных аналогично по индукции. Так как \(\overline{\Delta_i} \supset \overline{\Delta_{i + 1}}\), то у \(\{\overline{\Delta_i}\}\) сущестует общая точка \(z\). Запишем условие голоморфности в этой точке:
    \[f(z) - f(z_0) = f'(z_0)(z - z_0) + o(z - z_0), z \ra z_0\]
    Или:
    \[\forall \epsilon > 0 \exists \delta > 0: \forall z \in O_\delta(z_0) \subset D, \left| \frac{f(z) - f(z_0)}{z - z_0} - f'(z_0) \right| < \epsilon\]
    Получаем, что \(|f(z) - f(z_0) - f'(z_0)(z - z_0)| \le \epsilon|z - z_0|\). Заметим, что \(\int_{\partial \Delta_i} dz = 0, \int_{\partial \Delta_i} (z - z_0)dz = 0\), так как они обе имеют первообразную. Тогда:
    \[\left| \int_{\partial \Delta_i} f(z)dz \right| = \left| \int_{\partial \Delta_i} (f(z) - f(z_0) - f'(z_0)(z - z_0)) dz\right| \le \int_{\partial \Delta_i} \epsilon|z - z_0||dz| \le \epsilon \frac{l}{2^n}\int_{\partial \Delta_i}|dz| = \frac{\epsilon l^2}{4^n}\]
    Где \(l = \operatorname{Length} \partial\Delta\), тогда \(\operatorname{Length} \partial\Delta_n = \frac{l}{2^n}\). Заметим при этом, что \(\left| \int_{\partial \Delta} fdz \right| < \epsilon l^2\), получаем, что 
    \[\left|\int_{\partial \Delta_n} fdz\right| < \frac{\left| \int_{\partial \Delta} fdz\right|}{4^n}\]
    Но по построению, для каждого из треугольников \(\Delta_n\):
    \[\left|\int_{\partial \Delta_n} fdz\right| \ge \frac{\left| \int_{\partial \Delta} fdz\right|}{4^n}\]
    Получили противоречие.
\end{proof}

\begin{lemma}[Усиленная Лемма Гурса]
    Пусть \(f\) голоморфна в области \(D \setminus \{a\}\), \(f\) непрерывна в \(D\). Тогда \(\forall \overline{\Delta} \subset D: \int_{\partial \Delta} fdz = 0\). 
\end{lemma}
\begin{proof}
    \begin{enumerate}
        \item Пусть \(\overline{\Delta} \subset D, a \in D \setminus \overline{\Delta}\). В таком случае утверждение следует из леммы Гурса
        \item Пусть теперь \(a\) --- вершина \(\Delta\). Рассмотрим прямые, котоыре очень близко подходят к точке \(a\) и разбивают треугольник на четырехугольник и маленький треугольник. Положим \(\Delta_\lambda\) --- треугольник, и по лемме Гурса получим \(\int_{\partial (\Delta_ \setminus \Delta_\lambda)} fdz = 0\) (разобьем оставшийся четырехугольник на два треугольника). Заметим, что 
        \[\left| \int_{\partial \Delta} fdz \right| = \left| \int_{\partial \Delta_\lambda} fdz \right| \le \max_{\partial \Delta_\lambda}|f| \lambda \ra 0\]
        Получили желаемое
        \item Пусть теперь \(a\) лежит на стороне \(\Delta\). Тогда разбиваем его на два треугольника, в каждом из которых \(a\) будет вершиной
        \item Пусть теперь \(a\) лежит внутри \(\Delta\). Тогда разбиваем его на три треугольника, в каждом из которых \(a\) будет вершиной.
    \end{enumerate}
\end{proof}

\begin{lemma}[К теореме Коши]
    Пусть \(f\) непрерывна в области \(D\), \(D\) выпукла и \(\forall \overline{\Delta} \subset D: \int_{\partial \Delta} fdz = 0\). Тогда \(fdz\) --- полный дифференциал.
\end{lemma}
\begin{proof}
    Зафиксируем \(a \in D\) и положим \(F(z) = \int_{[a, z]}f(\zeta)d\zeta\). Заметим, что \(f(z_0)(z - z_0) = \int_{[z_0, z]}f(z_0)d\zeta\). Докажем, что \(F'(z) = f(z)\):
    \[\left| \frac{F(z) - F(z_0)}{z - z_0} - f(z_0) \right| \le \left| \frac{1}{z - z_0} \left( \underbrace{\int_{[a, z]}f(\zeta)d\zeta - \int_{[a, z_0]}f(\zeta)d\zeta}_{\int_{[z_0, z]}f(\zeta d\zeta)} - \int_{[z_0, z]}f(z_0)d\zeta\right)\right| = (*)\]
    Равенство внутри скобки получили из интегрирования по треугольнику с вершинами \(a, z, z_0\). В таком случае:
    \[(*) = \left| \frac{1}{z - z_0} \int_{[z_0, z]}(f(z) - f(z_0))d\zeta\right| \le \frac{\epsilon}{|z - z_0|}\int_{[z_0, z]}|dz| = \epsilon\]
\end{proof}

\begin{theorem}[Коши в выпуклой области]
    Пусть \(D\) --- выпуклая область в \(\Cm\), \(f\) голоморфна в \(D \setminus \{a\}\). Тогда \(fdz\) --- полный дифференциал, т.е. \(\int_\gamma fdz = 0\) по любой замнутой кусочно-гладкой кривой в \(D\).
\end{theorem}
\begin{proof}
    Следует из усиленной леммы Гурса и леммы выше.
\end{proof}

\subsection{Интегралы Коши}
\begin{definition}
    Пусть \(\rho\) непрерывна на  кусочной гладкой кривой \(\gamma\) в \(\Cm\). Тогда интеграл 
    \[\int_\gamma \frac{\rho(\zeta)d\zeta}{\zeta - z}, z \in \Cm \setminus \gamma\]
    Называется интегралом Коши.
\end{definition}

% \begin{note}
%     Функция \(\frac{1}{\zeta - z}\) называется ядром коши; таким образом, интеграл Коши --- свертка 
% \end{note}

\begin{theorem}
    Пусть \(\rho\) непрерывна на  кусочной гладкой кривой \(\gamma\) в \(\Cm\). Определим функцию:
    \[F_n(z, \rho) = \int_\gamma \frac{\rho(\zeta)d\zeta}{(\zeta - z)^n}\]
    Тогда \(F_n(z, \rho)\) голоморфна в \(\Cm \setminus \gamma\), причем \(F_n'(z, \rho) = nF_{n + 1}(z, \rho)\)
\end{theorem}
\begin{proof}
    Ведем индукцию по \(n\):
    \begin{enumerate}
        \item[] \textbf{База:} \(n = 1\). Докажем, что \(F_n(z, \rho)\) непрерывна в \(\Cm \setminus \gamma\). Пусть \(z_0 \in \Cm \setminus \gamma, \delta = \frac{1}{2}\operatorname{dist}(\gamma, z_0), z \in O_\delta(z_0) \setminus \Cm \setminus \gamma\). Тогда:
        \[F_1(z, \rho) - F_1(z_0, \rho) = (z - z_0)\int_\gamma \frac{\rho(\zeta)d\zeta}{(\zeta - z_0)(\zeta - z)}\]
        Положим \(M = \max_\gamma |\rho|\). Тогда:
        \[\left| \int_\gamma \frac{\rho(\zeta)d\zeta}{(\zeta - z_0)(\zeta - z)} \right| \le \int_\gamma \frac{|\rho(\zeta)||d\zeta|}{|\zeta - z_0||\zeta - z|} \le M \frac{1}{2\delta \cdot \delta} \int_\gamma |d \zeta| \le \frac{M_1}{2\delta^2}\]
        В таком случае функция \(F\) липшицева в \(O_\delta(z_0)\) с константой липшицевости \(\frac{M_1}{2\delta^2}\), в таком случае она является непрерывной. Теперь докажем голоморфность. Заметим, что:
        \[\frac{F_1(z, \rho) - F_1(z_0, \rho)}{z - z_0} = F_1\left( z, \frac{\rho(\zeta)}{\zeta - z_0} \right) \xrightarrow[z \ra z_0]{}F_1\left( z_0, \frac{\rho(\zeta)}{\zeta - z_0} \right) = F_2(z_0, \rho(\zeta))\]
        То есть \(F_1\) голоморфна в области \(D\).
        \item[] \textbf{Переход:} \(n - 1 \ra n\). Заметим, что:
        \[\frac{1}{(\zeta - z)^n} - \frac{1}{(\zeta - z_0)^n} = \frac{1}{(\zeta - z)^{n - 1}}\left( \frac{1}{\zeta - z} - \frac{1}{\zeta - z_0} \right) + \frac{1}{\zeta - z_0}\left( \frac{1}{(\zeta - z)^{n - 1}} - \frac{1}{(\zeta - z_0)^{n - 1}} \right)\]
        Тогда:
        \[F_n(z, \rho) - F_n(z_0, \rho) = \int_\gamma \frac{\rho(\zeta)(z - z_0)d\zeta}{(\zeta - z)^n(\zeta - z_0)} + \int_\gamma \frac{\rho(\zeta)}{(\zeta - z_0)}\left( \frac{1}{(\zeta - z)^{n - 1}} - \frac{1}{(\zeta - z_0)^{n - 1}} \right)d\zeta =\]
        \[= (z - z_0)F_n\left( z, \frac{\rho(\zeta)}{\zeta - z_0} \right) + \left( F_{n - 1}\left(z, \frac{\rho(\zeta)}{\zeta - z_0}\right) - F_{n - 1}\left( z_0, \frac{\rho(\zeta)}{\zeta - z_0} \right) \right)\]
        Непрерывность \(F_n\) доказывается аналогично базе. Для дифференцируемости точно так же запишем:
        \[\frac{F_n(z, \rho) - F_n(z_0, \rho)}{z - z_0} = F_n\left( z, \frac{\rho(\zeta)}{\zeta - z_0} \right) + \frac{F_n\left(z, \frac{\rho(\zeta)}{\zeta - z_0}\right) - F_n\left(z_0, \frac{\rho(\zeta)}{\zeta - z_0}\right)}{z - z_0}\]
        Переходим к пределу при \(z \ra z_0\):
        \[F_n'(z_0, \rho) = F_n\left( z_0, \frac{\rho(\zeta)}{\zeta - z_0} \right) + F_{n - 1}'\left( z_0, \frac{\rho(\zeta)}{\zeta - z_0} \right) = \]
        \[= F_{n + 1}(z_0, \rho) + (n - 1)F_n\left( z_0, \frac{\rho(\zeta)}{\zeta - z_0} \right) = nF_{n + 1}(z_0, \rho)\]
    \end{enumerate}
\end{proof}

\begin{corollary}
    Пусть \(\gamma\) --- кусочно-гладкая кривая, \(z \in \Cm \setminus \gamma\) и \(F(z)\) --- интеграл Коши:
    \[F(z) = \int_\gamma \frac{\rho(\zeta)d\zeta}{\zeta - z}, z \in \Cm \setminus \gamma\]
    Тогда \(F(z)\) бесконечно дифференцируема в \(\Cm \setminus \gamma\), причем:
    \[F^{(n)}(z) = n!\int_{\gamma} \frac{\rho(\zeta)d\zeta}{(\zeta - z)^{n + 1}}\]
\end{corollary}

\begin{theorem}[Интегральная формула Коши в круге]
    Пусть \(f\) голоморфна в области \(D \supset \overline{O_r(a)}\). Тогда \(\forall z \in O_r(a)\) верно:
    \[f(z) = \frac{1}{2\pi i} \int_{\gamma_r}\frac{f(\zeta)d\zeta}{\zeta - z}\]
    Где \(\gamma_r = \partial O_r(a)\) при обходе против часовой стрелки.
\end{theorem}
\begin{proof}
    Положим \(R > r\) --- такое, что \(O_R(a) \subset D\). Тогда при \(z \in O_r(a), \zeta \in O_R(a)\) пололжим:
    \[g(\zeta, z) = \left\{\begin{array}{l}
        \frac{f(\zeta) - f(z)}{\zeta - z}, z \ne \zeta \\
        f'(z), z = \zeta
    \end{array}\right.\]
    Докажем, что \(\int_{\gamma_r} g(\zeta, z)d\zeta = 0\). Это равносильно тому, что:
    \[\int_{\gamma_r} \frac{f(\zeta)d\zeta}{\zeta - z} = f(z)\underbrace{\int_{\gamma_r}\frac{d\zeta}{\zeta - z}}_{h(z)}\]
    По свойству интеграла Коши, получаем: \(h'(z) = \int_{\gamma_r}\frac{d\zeta}{(\zeta - z)^2} = 0\) и тогда \(h(z) = h(a) = \frac{1}{2\pi i}\).
\end{proof}

\begin{corollary}
    Пусть функция \(f\) голоморфна в области \(D \supset \overline{O_r(a)}\). Тогда \(f\) бесконечно дифференцируема в области \(D\) и \(f^{(n)}(z) = \frac{n!}{2\pi i}\int_{\gamma_r}\frac{f(\zeta)d\zeta}{(\zeta - z)^{n + 1}}, z \in O_r(a)\).
\end{corollary}

\begin{theorem}[Мореры]
    Пусть \(f\) непрерывна в области \(D\) и \(\forall \overline{\Delta} \subset D\) выполнено \(\int_{\partial D}fdz = 0\). Тогда \(f\) голоморфна в области \(D\)
\end{theorem}
\begin{proof}
    По теорме Коши, \(f = F'\) в окрестности любой точки \(z_0\). Но так как любая голоморфная функция бесокненчо дифференцируема, то \(f\) --- голоморфна
\end{proof}
