\section{Метрические и топологические пространства}

\begin{definition}
    Пара \((X, \rho)\) называется метрическим пространством (\(X \ne \emptyset, \rho: X \times X \ra \R\)), если
    \begin{enumerate}
        \item \(\rho(x, y) \ge 0\), причем равенство достигается только в том случае, когда \(x = y\).
        \item \(\rho(x, y) = \rho(y, x)\)
        \item \(\rho(x, y) \le \rho(x, z) + \rho(z, y)\)
    \end{enumerate}
\end{definition}

\subsection{Словарь используемых терминов}

Пусть \(X\) --- произвольное метрическое пространство.

\begin{enumerate}
    \item Подпространство --- любое подмножество \(X\)
    
\subsubsection{География}
    \item Ограниченное множество \(Y \subset X\) --- такое множество, что \(\sup \rho_{x, y \in Y}(x, y) < +\infty \Lra Y \subset B(x, r)\) для некоторого \(x\)
    \item \(\rho(x, Y) = \inf_{y \in Y} \rho(x, y)\)
    \item Открытый шар \(B(x, r) = \{y: \rho(y, x) < r\}\), замкнутый шар \(\overline{B}(x, r) = \{y: \rho(y, x) \le r\}\)

\subsubsection{Замкнутость}
    \item Точка \(x\) является точкой прикосновения множества \(Y\), если выполнено одно из равносильных условий:
    \begin{itemize}
        \item \(\forall B(x) \cap Y \ne \emptyset\)
        \item \(\rho(x, Y) = 0\)
        \item \(\exists \{x_n\} \subset Y: x_n \ra x\) 
    \end{itemize}
    \item Замыкание множества (\(\overline{Y}\)) --- объединение множества со всеми точками прикосновения (или, что эквивалентно, просто множество всех точек прикосновения).
    \item Замкнутое множество --- такое, что \(\overline{Y} = Y\)
    
\subsubsection{Открытость}
    \item Точка \(x\) называется внутренней для множества \(Y\), если \(\exists B(x) \subset Y\)
    \item Внутренность множества (или открытое ядро) --- \(\stackrel{\circ}{Y} = Y^{\circ} = \operatorname{int}Y\) --- множество всех внутренних точек
    \item Открытое множество --- такое, что \(Y^{\circ} = Y\).


    \item Сходящаяся последовательность --- \(x_n \ra x\), если \(\forall B(x) \exists N: \forall n > N: x_n \in B(x)\).
    
    Топология --- совокупность всех открытых множеств

\subsubsection{Прочее}
    \item \(A\) называется плотным в \(B\), если \(\overline{A} = B\)
    \item \(A\) всюду плотно, если \(\overline{A} = X\)
    \item \(A\) нигде не плотно \(\Lra\) не плотно ни в одном шаре
    
    Представим мышей, которые скушали головку сыра, да так, что другие мыши, которые пришли, в каждом шаре не нащли, что бы им поесть. Ньюанс заключается в том, что дети едят так называемую сладкую вату, которая сама по себе нигде не плотна, и получают удовольствие, но пример понятен

    \item \(X\) --- сепарабельное пространство, если \(\exists A\) --- не более, чем счетное, такое, что \(\overline{A} = X\)
    
    \item \(X\) --- несвязно, если \(X = G_1 \sqcup G_2\), где \(G_1, G_2 \ne \emptyset\) --- открытые. Соответственно, \(X\) --- связно, если оно не является несвязным.
    
    Вообще говоря, есть еще понятие линейной связности. У гуманитариев аналогом ему является слово ''нарратив''

    \item База топологии --- система открытых множеств \(\mathcal{B}\) называется базой, если \(\forall G\) --- открытого, \(G = \bigcup B_\alpha, B_\alpha \in \mathcal{B}\)
    \item Сравнение топологий --- \(\tau_1 \prec \tau_2 \Lra \tau_1 \subset \tau_2\)
    \item \(CY = \overline{Y} = X \setminus Y\), где \(Y \subset X\) --- дополнение к множеству \(Y\).
\end{enumerate}

\begin{theorem}
    Пусть \(X\) --- метрическое пространство. \(G_1 \subset X\) --- открытое \(\Lra F = X \setminus G = CG\) --- замкнутое.
\end{theorem}
\begin{proof}
    \begin{enumerate}
        \item[\(\Ra\)] Покажем, что если \(x \notin F \Ra x\) --- не точка прикосновения. \(x \notin F \Lra x \in G \Ra \exists B(x) \subset G \Ra \exists B(x): B(x) \cap F = \emptyset\)
        \item[\(\La\)] предоставляется читателю в качестве нетрудного упражнения
    \end{enumerate}
\end{proof}

    \begin{enumerate}
        \item \(\bigcup G_\alpha \in \tau_X\)
        \item \(\bigcap_{i = 1}^n G_{\alpha_n} \in \tau_X\)

\begin{definition}
    Топологическое пространство --- \((X, \tau)\), где \(\tau \subset 2^X\), если:
    \begin{enumerate}
        \item \(\emptyset, X \in \tau\)
        \item Для \(\forall \{G_\alpha\} \subset \tau\):
        \begin{itemize}
            \item \(\bigcup G_\alpha \in \tau\)
            \item \(\bigcap_{i = 1}^n G_{\alpha_n} \in \tau\)
        \end{itemize}
    \end{enumerate}
\end{definition}

\begin{example}[Дискретная метрика]
    Рассмотрим \((X, \rho): \rho(x, y) = \left\{\begin{array}{l}
        1, x \ne y \\
        0, x = y
    \end{array}\right.\). Заметим, что \(\tau_X = 2^X\)
\end{example}

\begin{theorem}
    Пусть \(X\) --- метрическое пространство. Тогда:
    \begin{enumerate}
        \item \(B(x, \epsilon)\) --- открытое множество
        \item \(\operatorname{int}M\) --- открытое множество
        \item \(\overline{B}(x, \epsilon)\) --- замкнутое множество
        \item \(\overline{M}\) --- замкнутое множество
    \end{enumerate}
\end{theorem}
\begin{proof}
    \begin{enumerate}
        \item Пусть \(\rho(y, x) = r < \epsilon\). В таком случае для любого \(z \in B\left( y, \frac{\epsilon - r}{2} \right)\) верно \(\rho(y, z) \le \rho(y, x) + \rho(x, z) < \epsilon\)
        \item Возьмем произвольную точку из \(\operatorname{int}M\). Заметим, что \(B(x, \epsilon) \subset M\) (из определения \(\operatorname{int}M\)). Тогда \(\operatorname{int}B(x, \epsilon) \subset \operatorname{int}M \Ra B(x, \epsilon) = \operatorname{int}B(x, \epsilon) \subset \operatorname{int}M\).
        \item[3, 4.] Предоставляется читателю в качестве нетрудного упражнения
    \end{enumerate}
\end{proof}

\subsection{Непрерывные отображения топологических пространств}
Пусть \(f: X \ra Y\) --- отображение метрических пространств.
\begin{definition}[Непрерывность в МП по Коши]
    \(f\) называется непрерывной в точке \(x_0\), если \(\forall V(f(x_0)) \exists U(x_0): f(U) \subset V\)
\end{definition}

\begin{definition}[Непрерывность в МП по Гейне]
    \(f\) называется непрерывной в точке \(x_0\), если \(\forall x_n \ra_X x_0 \Ra f(x_n) \ra_Y f(x_0)\)
\end{definition}

\begin{note}
    На матане мы доказывали, что эти определения эквивалентны.
\end{note}

\begin{definition}[Непрерывность для топологических пространств]
    Пусть \(f: X \ra Y\) --- отображение топологических пространств. \(f\) называется непрерывной в точке \(x_0\), если \(\forall V(f(x_0)) \exists U(x_0): f(U) \subset V\) (определение абсолютно совпадает с определением по Коши для МП).
\end{definition}

\begin{theorem}
    Пусть \(X, Y\) --- ТП, \(f: X \ra Y\). Следующие утверждения эквивалентны:
    \begin{enumerate}
        \item \(f\) --- непрерывна
        \item \(\forall G \in \tau_Y: f^{-1}(G) \in \tau_X\)
        \item \(\forall\) замкнутого \(F \subset Y: f^{-1}(F)\) замкнуто в \(X\).
    \end{enumerate}
\end{theorem}
\begin{proof}\indent
    \begin{enumerate}
        \item[\((1) \Ra (2)\)] Пусть \(G \in \tau_Y\). Рассмотрим \(H = f^{-1}(G)\). По определению, \(\forall f(x) \in G \exists H_x \in \tau_X \ni x: H \supset H_x\). Заметим, что \(\bigcup_{x \in H} H_x \subset H \Ra \bigcup_{x \in H} H_x = H \Ra H \in \tau_H\).
        \item[\((2) \Ra (1)\)] Рассмотрим \(V(f(x_0)) = G\). В таком случае \(\exists U(x_0) = f^{-1}(G) \in \tau_Y\).
        \item[\((2) \Lra (3)\)] Очевидно следует из \(f^{-1}(CF) = Cf^{-1}(F)\)
    \end{enumerate}
\end{proof}

\begin{definition}(Гомеоморфизм ТП)
    Пусть \(X, Y\) --- ТП. Они называются гомеоморфными, если \(\exists f: X \ra Y\) --- биекция, такая что \(f, f^{-1}\) --- непрерывны.
\end{definition}

\end{exercise}

\section{Полные метрические пространства}

\begin{definition}
    Метрическое пространство \(X\) называется полным (ПМП), если любая фундаментальная последовательность элементов из \(X\) является сходящейся.
\end{definition}

\begin{example}[Полные МП]
    \(\R^n, [0, 1], C[a, b]\)
\end{example}

\begin{example}[Неполные МП]
    \(\Q, (0, 1), C_1[a, b]\) --- множество \(C[a, b]\) с нормой \(\|f\| = \int_a^b |f(x)|dx\)
\end{example}

\begin{theorem}
    \(X\) --- полное метрическое пространство \(\Lra \forall \{\overline{B}(x_n, r_n)\}\) --- последовательность замкнутых вложенных шаров, таких, что \(r_n \ra 0\), верно \(\exists ! x \in \bigcap \overline{B}_n(x_n, r_n)\)
\end{theorem}
\begin{proof}
    \begin{enumerate}
        \item[\(\Ra\)] Рассмотрим последовательность \(\{x_n\}\). Она фундаментальная, так как \(\rho(x_n, x_m) \le \min(r_m, r_n)\). В таком случе \(\lim_{n \ra \infty} x_n = x\). Заметим, что \(x_m \in \overline{B}(x_n, r_n)\) для \(m > n\) (из вложенности). Тогда из замкнутости получаем: \(\lim_{m \ra \infty} x_m \in \overline{B}(x_n, r_n)\), т.е. \(x \in x \in \bigcap \overline{B}_n(x_n, r_n)\). Докажем, что \(x\) единственна. Пусть \(x_1 \ne x_2 \in \bigcap \overline{B}_n(x_n, r_n)\). Заметим, что для достаточно большого \(N\) получем: \(r_N < \frac{\rho(x_1, x_2)}{2}\). В таком случае, либо \(x_1 \notin \overline{B}(x_N, r_N)\), либо \(x_2 \notin \overline{B}(x_N, r_N)\).
        \item[\(\La\)] Рассмотрим фундаментальную последовательность \(\{x_n\}\). Требуется доказать ее сходимость. Положим \(r_k = \frac{1}{k}, y_k = x_{n_k}\), где \(n_k\) --- такое, что \(\forall m > n_k: \rho(x_m, x_{n_k}) < r_k\) (существует из определения фундаментальности). Заметим, что при \(k \ra \infty\), \(n_k \ra \infty\). В таком случае, нашли последовательность шаров \(\overline{B}(x_{n_k}, r_k)\), удовлетворяющих утверждению теоремы. Тогда, \(\exists x \in \bigcap \overline{B}(x_{n_k}, r_k)\), и \(x_{n_k} \ra x\). Далее, из фундаментальности и полученной сходимости, имеем \(x_n \ra x\), что и требовалось.
    \end{enumerate}
\end{proof}

\begin{proposition}
    По сути, утверждение выше эквивалентно следующему: если ягода большая, с зеленой коркой и красной мякотью, то это --- арбуз. Заметим, также, что верно и обратное.
\end{proposition}

\begin{note}
    Заметим, что в условиях теоремы нельзя отказаться от свойства замкнутости шара. Аналогично, нельзя отказаться от вложенности. Но можно ли отказаться от \(r_n \ra 0\)? Оказывается, что нет.
\end{note}

\begin{example}
    Рассмотрим \(X = \N, \rho(m, n) = 1 + \frac{1}{m + n}\). Рассмотрим \(x_n = n, r_n = \frac{1}{2n}\). В таком случае \(\overline{B}(x_n, r_n) = \{m \ge n, m \in \N\}\). Однако, \(\bigcap \overline{B}(x_n, r_n) = \emptyset\).
\end{example}

\begin{definition}
    Пусть \(X\) --- метрическое пространство. \(X\) называется множеством второй категории, если \(X\) нельзя представить в виде \(X = \bigcup_{n = 1}^\infty M_n\), где \(\forall n: M_n\) --- нигда не плотное множество
\end{definition}

\begin{theorem}[Бэр]
    Если \(X\) --- полное метрическое пространство, то \(X\) --- нигде не плотное множество.
\end{theorem}
\begin{proof}
    Предположим противное. Пусть \(X = \bigcup M_n\), где \(\forall n: M_n\) --- нигда не плотное множество. Возьмем \(\forall \overline{B}_0(x_0, 1)\). По индукции построим следующую последовательности \(\{x_n\}, \{r_n\} \) такие, что \(\overline{B}(x_n, r_n) \in \overline{B}(x_{n - 1}, r_{n - 1}) \), и \(\overline{B}(x_n, r_n) \cap M_n = \emptyset\) (такие можно взять из определений нигде не плотных множеств \(M_n\)), и \(r_n \searrow 0\). В таком случае получаем \(x \in \bigcap_{n = 1}^\infty \overline{B}(x_n, r_n)\). Тогда \(x \notin M_n \forall n\), получили противоречие
\end{proof}

\begin{theorem}[Банах, 1922]
    Пусть \(X\) --- полное метрическое пространство, \(f: X \ra X\) --- сжимающее, т.е. \(\exists \alpha \in (0, 1): \rho(fx, fy) \le \alpha \rho(x, y)\). Тогда \(\exists! \) неподвижная точка отображения \(f\), т.е. \(\exists! x \in X: fx = x\)
\end{theorem}
\begin{proof}
    Рассмотрим \(x_n = f^nx_0\) --- фунаментальная последовательность. Действительно, 
    \[\rho(x_n, x_m) \le \alpha^n \rho(x_0, f^{m - n}x_0) \le (\rho(x_0, x_1) + \dots + \rho(x_{m - n - 1}, x_{m - n})) \cdot \sum_{k = n}^{m - 1} \alpha^k \le\]
    \[\le \rho(x_0, fx_0) \cdot \sum_{k = n}^\infty \alpha^k \ra 0\quad(n \ra \infty)\]
    Тогда \(\exists x = \lim_{n \ra \infty} x_n\). Заметим, что \(f\) --- непрерывное отображение, поэтому \(fx_n \ra fx\). При этом \(fx_n = x_{n + 1} \ra x\). Получили, что \(fx = x\). Единственность доказывается аналогично теореме Бэра.
\end{proof}
